# Albums Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Personal, shareable albums in the gallery: curated from channel files plus direct uploads, private or server-visible, with collaborators.

**Architecture:** A new Go package `server/internal/albums` owns the `albums`, `album_members` and `album_items` tables, access rules, the HTTP API and events. Attachments gain an optional `album_id` as an alternative home to `channel_id`; the `files` package stores album uploads and authorizes their downloads through a small `Albums` interface that the albums service implements. Album file listings reuse the gallery's `FileItem` shape, hydrated by a new exported `messages.Service.FileItems`. The web client adds an Albums view and an album scope to the existing gallery.

**Tech Stack:** Go 1.x, huma v2, pgx v5, goose migrations, Postgres; TypeScript, React 18, react-router, zustand, shadcn/ui (Radix) + Tailwind v4, vitest + msw, Playwright.

**Spec:** `docs/superpowers/specs/2026-10-12-albums-design.md`

## Global Constraints

- Migration file: `server/internal/db/migrations/00039_albums.sql` (00038 is `audit_log`).
- Limits: 100 albums per owner, 5,000 items per album, 25 collaborators per album; codes `ALBUMS_FULL`, `ALBUM_FULL`, `COLLABORATORS_FULL` (400).
- Album title: 1-100 characters. `visibility` is `"private"` or `"server"`.
- Unknown or invisible album: `404 NOT_FOUND` "album not found". No rights: `403 MISSING_PERMISSION` (`httpx.MissingPermission()`).
- Every album check requires server membership: `communities.BasePerms(ctx, user) != 0`.
- Album uploads never appear in `/api/files`, search, channels, or playlists.
- Deleting a message removes its files from albums; deleting an album deletes its direct uploads.
- Web: dark-only, no `dark:` classes, tokens only, `cn()` from `@/lib/utils`, imports via `@/`. `Button` defaults to `type="button"`.
- Accessible names (e2e selects by them): "Albums", "All files", "New album", "Add to album", "Remove from album", "Album settings", "Who can see it", "Leave album", "Delete album".
- Never commit or push: AGENTS.md says the user commits. Each task ends by staging and proposing a commit message, not committing.

## Review Focus

- **A collaborator removed from the server** keeps an `album_members` row; they must get 404 on the album and its files, and album uploads must 404 on download. (Task 2, Task 3 tests.)
- **An album upload's download URL** shared with someone who can't see the album must 404, including after the album turns private. (Task 3 test.)
- **The 24-hour unsent-upload sweep** must never delete album uploads. (Task 1 test.)
- **A channel file the viewer can't read** must not leak through the album listing or the album card's cover; only the count includes it. (Task 3 `TestAlbumChannelFiles`; the cover query in Task 2 uses the same readable-channel filter, so review it against the listing condition.)
- **Gallery code that assumed `item.message`** (tiles, panel, player, menus) must not crash on album uploads with no message. (Task 5 tests and typecheck.)

```sql
-- +goose Up
-- Albums: personal collections of files, from channels or uploaded into the album.
CREATE TABLE albums (
    id         BIGINT PRIMARY KEY, -- snowflake; created_at is its time
    owner_id   BIGINT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
    title      TEXT NOT NULL,
    visibility TEXT NOT NULL DEFAULT 'private' CHECK (visibility IN ('private', 'server')),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX albums_owner_idx ON albums (owner_id);

-- Collaborators; the owner is not listed.
CREATE TABLE album_members (
    album_id BIGINT NOT NULL REFERENCES albums (id) ON DELETE CASCADE,
    user_id  BIGINT NOT NULL REFERENCES users (id) ON DELETE CASCADE,
    PRIMARY KEY (album_id, user_id)
);
CREATE INDEX album_members_user_idx ON album_members (user_id);

-- An attachment lives in a channel or, uploaded straight into one, in an album.
ALTER TABLE attachments
    ALTER COLUMN channel_id DROP NOT NULL,
    ADD COLUMN album_id BIGINT REFERENCES albums (id) ON DELETE CASCADE,
    ADD CONSTRAINT attachments_home_check CHECK ((channel_id IS NULL) <> (album_id IS NULL));
CREATE INDEX attachments_album_idx ON attachments (album_id) WHERE album_id IS NOT NULL;

CREATE TABLE album_items (
    album_id      BIGINT NOT NULL REFERENCES albums (id) ON DELETE CASCADE,
    attachment_id BIGINT NOT NULL REFERENCES attachments (id) ON DELETE CASCADE,
    added_by      BIGINT REFERENCES users (id) ON DELETE SET NULL,
    added_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    PRIMARY KEY (album_id, attachment_id)
);
CREATE INDEX album_items_attachment_idx ON album_items (attachment_id);

-- +goose Down
DROP TABLE album_items;
DELETE FROM attachments WHERE album_id IS NOT NULL;
ALTER TABLE attachments
    DROP CONSTRAINT attachments_home_check,
    DROP COLUMN album_id,
    ALTER COLUMN channel_id SET NOT NULL;
DROP TABLE album_members;
DROP TABLE albums;
```

```go
// FileItems hydrates listed files, in order: each with its message and channel, or (album uploads)
// without. Files deleted since they were listed are left out.
func (s *Service) FileItems(ctx context.Context, hits []FileHit) ([]FileItem, error) {
	out := []FileItem{}
	if len(hits) == 0 {
		return out, nil
	}
	attIDs := make([]id.ID, len(hits))
	for i, h := range hits {
		attIDs[i] = h.AttachmentID
	}
	atts, err := s.attachments.ForIDs(ctx, attIDs)
	if err != nil {
		return nil, err
	}
	msgIDs := []id.ID{}
	seen := map[id.ID]bool{}
	for _, a := range atts {
		if m := a.MessageID(); m != nil && !seen[*m] {
			seen[*m] = true
			msgIDs = append(msgIDs, *m)
		}
	}
	byID := map[id.ID]Message{}
	refs := map[id.ID]ChannelRef{}
	if len(msgIDs) > 0 {
		mrows, err := s.db.Query(ctx, `SELECT `+messageCols+` FROM messages WHERE id = ANY($1)`, id.Int64s(msgIDs))
		if err != nil {
			return nil, err
		}
		ms, err := collectMessages(mrows)
		if err != nil {
			return nil, err
		}
		if err := s.hydrate(ctx, ms); err != nil {
			return nil, err
		}
		for _, m := range ms {
			byID[m.ID] = m
		}
		if refs, err = s.channelRefs(ctx, channelIDsOf(ms)); err != nil {
			return nil, err
		}
	}
	for _, h := range hits {
		a, ok := atts[h.AttachmentID]
		if !ok {
			continue // deleted between the two queries
		}
		it := FileItem{Attachment: a, Kind: h.Kind, UploaderID: a.UploaderID(), CreatedAt: a.ID.Time()}
		if mid := a.MessageID(); mid != nil {
			m, ok := byID[*mid]
			if !ok || m.Deleted {
				continue
			}
			ref := refs[m.ChannelID]
			it.Message, it.Channel = &m, &ref
		}
		out = append(out, it)
	}
	return out, nil
}
```

```go
// access loads an album the caller can see and says how: owner, collaborator or viewer.
// Anything else is 404, so private albums don't reveal that they exist.
func (s *Service) access(ctx context.Context, userID, albumID id.ID) (*row, string, error) {
	p, err := s.base(ctx, userID)
	if err != nil {
		return nil, "", err
	}
	a, err := scanRow(s.db.QueryRow(ctx, `SELECT `+rowCols+` FROM albums a WHERE a.id = $1`, albumID))
	if errors.Is(err, pgx.ErrNoRows) || (err == nil && p == 0) {
		return nil, "", httpx.NotFound("album")
	}
	if err != nil {
		return nil, "", err

	case a.ownerID == userID:
		return a, RoleOwner, nil
	case a.has(userID):
		return a, RoleCollaborator, nil
	case a.visibility == "server":
		return a, RoleViewer, nil
	}
	return nil, "", httpx.NotFound("album")
}

// list returns the albums userID can see, most recently changed first; with contains set, each says
// whether that attachment is in it.
func (s *Service) list(ctx context.Context, userID id.ID, contains *id.ID) ([]Album, error) {
	out := []Album{}
	p, err := s.base(ctx, userID)
	if err != nil || p == 0 {
		return out, err
	}
	scope, err := s.channels.Readable(ctx, userID)
	if err != nil {
		return nil, err
	}
	var containsArg *int64
	if contains != nil {
		c := int64(*contains)
		containsArg = &c
	}
	rows, err := s.db.Query(ctx, `SELECT `+rowCols+`,
		(SELECT count(*) FROM album_items i WHERE i.album_id = a.id),
		(SELECT t.id FROM album_items i JOIN attachments t ON t.id = i.attachment_id
			WHERE i.album_id = a.id AND t.inline AND `+fmt.Sprintf(readableSQL, "$2")+` ORDER BY t.id DESC LIMIT 1),
		$3::bigint IS NOT NULL AND EXISTS (SELECT 1 FROM album_items i WHERE i.album_id = a.id AND i.attachment_id = $3)
		FROM albums a
		WHERE a.visibility = 'server' OR a.owner_id = $1
			OR EXISTS (SELECT 1 FROM album_members m WHERE m.album_id = a.id AND m.user_id = $1)
		ORDER BY a.updated_at DESC, a.id DESC`, userID, id.Int64s(scope), containsArg)
	if err != nil {
		return nil, err
	}
	type listed struct {
		row
		count int
		cover *int64
		has   bool
	}
	var got []listed
	for rows.Next() {
		var l listed
		var members []int64
		if err := rows.Scan(&l.id, &l.ownerID, &l.title, &l.visibility, &l.updatedAt, &members, &l.count, &l.cover, &l.has); err != nil {
			rows.Close()
			return nil, err
		}
		for _, m := range members {
			l.memberIDs = append(l.memberIDs, id.ID(m))
		}
		got = append(got, l)
	}
	rows.Close()
	if err := rows.Err(); err != nil {
		return nil, err
	}
	var coverIDs []id.ID
	for _, l := range got {
		if l.cover != nil {
			coverIDs = append(coverIDs, id.ID(*l.cover))
		}
	}
	covers, err := s.files.ForIDs(ctx, coverIDs)
	if err != nil {
		return nil, err
	}
	for _, l := range got {
		a := toAlbum(&l.row, l.count, roleOf(&l.row, userID))
		if l.cover != nil {
			if c, ok := covers[id.ID(*l.cover)]; ok {
				a.Cover = &c
			}
		}
		if contains != nil {
			has := l.has
			a.Contains = &has
		}
		out = append(out, a)
	}
	return out, nil
}

		return err

			rows.Close()
			return err
		}
		sums = append(sums, sum)
		if thumb != "" {
			sums = append(sums, thumb)
		}
	}
	rows.Close()
	if err := rows.Err(); err != nil {
		return err
	}
	if _, err := tx.Exec(ctx, `DELETE FROM albums WHERE id = $1`, albumID); err != nil { // cascades to members, items, uploads
		return err

		if err := audit.Record(ctx, tx, audit.AlbumDelete, albumID, audit.LogDetails{Name: a.title}); err != nil {
			return err
		}
	}
	if err := tx.Commit(ctx); err != nil {
		return err
	}
	s.publishDelete(ctx, a)
	return s.files.RemoveBlobs(ctx, sums)
}

// audience is who hears about an album: the whole server, or its owner and collaborators.
func (s *Service) audience(ctx context.Context, visibility string, ownerID id.ID, memberIDs []id.ID) events.Scope {
	if visibility == "server" {
		if sid, err := s.members.ServerID(ctx); err == nil {
			return events.Scope{CommunityID: sid}
		}
	}
	return events.Scope{UserIDs: append([]id.ID{ownerID}, memberIDs...)}
}

// publishUpdate sends ALBUM_UPDATE to its audience, and ALBUM_DELETE to anyone who could see it
// before (before may be nil) and no longer can.
func (s *Service) publishUpdate(ctx context.Context, a *Album, before *row) {
	if before != nil {
		lost := []id.ID{}
		for _, m := range before.memberIDs {
			if !contains(a.MemberIDs, m) {
				lost = append(lost, m)
			}
		}
		switch {
		case before.visibility == "server" && a.Visibility == "private":
			s.bus.Publish(events.Event{Type: "ALBUM_DELETE", Data: AlbumDeleted{ID: a.ID}, Scope: s.audience(ctx, "server", a.OwnerID, nil)})
		case a.Visibility == "private" && len(lost) > 0:
			s.bus.Publish(events.Event{Type: "ALBUM_DELETE", Data: AlbumDeleted{ID: a.ID}, Scope: events.Scope{UserIDs: lost}})
		}
	}
	shown := *a
	shown.Role, shown.Contains = "", nil
	s.bus.Publish(events.Event{Type: "ALBUM_UPDATE", Data: shown, Scope: s.audience(ctx, a.Visibility, a.OwnerID, a.MemberIDs)})
}

func (s *Service) publishDelete(ctx context.Context, r *row) {
	s.bus.Publish(events.Event{Type: "ALBUM_DELETE", Data: AlbumDeleted{ID: r.id}, Scope: s.audience(ctx, r.visibility, r.ownerID, r.memberIDs)})
}

func (s *Service) publishItems(ctx context.Context, albumID id.ID) {
	var visibility string
	var ownerID id.ID
	var members []int64
	err := s.db.QueryRow(ctx, `SELECT a.visibility, a.owner_id, ARRAY(SELECT user_id FROM album_members WHERE album_id = a.id)
		FROM albums a WHERE a.id = $1`, albumID).Scan(&visibility, &ownerID, &members)
	if err != nil {
		return
	}
	ids := make([]id.ID, len(members))
	for i, m := range members {
		ids[i] = id.ID(m)
	}
	s.bus.Publish(events.Event{Type: "ALBUM_ITEMS_UPDATE", Data: ItemsChanged{AlbumID: albumID}, Scope: s.audience(ctx, visibility, ownerID, ids)})
}

func contains(ids []id.ID, x id.ID) bool {
	for _, v := range ids {
		if v == x {
			return true
		}
	}
	return false
}
```

```go
package albums

import (
	"context"
	"errors"
	"fmt"
	"net/http"
	"strconv"
	"strings"

	"github.com/danielgtaylor/huma/v2"
	"github.com/jackc/pgx/v5"

	"github.com/datrope/datrope/server/internal/httpx"
	"github.com/datrope/datrope/server/internal/id"
	"github.com/datrope/datrope/server/internal/messages"
)

var likeEscaper = strings.NewReplacer(`\`, `\\`, `%`, `\%`, `_`, `\_`)

var fileKinds = map[string]bool{"image": true, "video": true, "audio": true, "document": true, "archive": true, "other": true}

type filesQuery struct {
	kinds    []string
	uploader *id.ID
	text     string
	before   *id.ID
	limit    int
}

// files lists an album's files the caller can see, newest upload first.
func (s *Service) files(ctx context.Context, userID, albumID id.ID, fq filesQuery) (*messages.FilesPage, error) {
	if _, _, err := s.access(ctx, userID, albumID); err != nil {
		return nil, err
	}
	scope, err := s.channels.Readable(ctx, userID)
	if err != nil {
		return nil, err
	}
	var args []any
	add := func(v any) string {
		args = append(args, v)
		return "$" + strconv.Itoa(len(args))
	}
	al := add(albumID)
	sc := add(id.Int64s(scope))
	// Attachments are aliased a, as messages.KindSQL expects. A file is shown when it is this
	// album's upload, or a sent, undeleted file in a channel (or a thread of one) the caller reads.
	conds := []string{
		"i.album_id = " + al,
		"(a.album_id = " + al + " OR (m.id IS NOT NULL AND NOT m.deleted AND (a.channel_id = ANY(" + sc + ")" +
			" OR (c.type = 'thread' AND c.parent_channel_id = ANY(" + sc + ")))))",
	}
	if len(fq.kinds) > 0 {
		conds = append(conds, messages.KindSQL+" = ANY("+add(fq.kinds)+")")
	}
	if fq.uploader != nil {
		conds = append(conds, "a.uploader_id = "+add(*fq.uploader))
	}
	if fq.text != "" {
		p := add(fq.text)
		conds = append(conds, "(a.filename ILIKE '%' || "+add(likeEscaper.Replace(fq.text))+" || '%'"+
			" OR to_tsvector('simple', coalesce(m.content, '')) @@ plainto_tsquery('simple', "+p+"))")
	}
	if fq.before != nil {
		conds = append(conds, "a.id < "+add(*fq.before))
	}
	rows, err := s.db.Query(ctx, `SELECT a.id, `+messages.KindSQL+`
		FROM album_items i JOIN attachments a ON a.id = i.attachment_id
		LEFT JOIN messages m ON m.id = a.message_id LEFT JOIN channels c ON c.id = a.channel_id
		WHERE `+strings.Join(conds, " AND ")+` ORDER BY a.id DESC LIMIT `+add(fq.limit+1), args...)
	if err != nil {
		return nil, err
	}
	var hits []messages.FileHit
	for rows.Next() {
		var h messages.FileHit
		if err := rows.Scan(&h.AttachmentID, &h.Kind); err != nil {
			rows.Close()
			return nil, err
		}
		hits = append(hits, h)
	}
	rows.Close()
	if err := rows.Err(); err != nil {
		return nil, err
	}
	page := &messages.FilesPage{Items: []messages.FileItem{}}
	if len(hits) > fq.limit {
		hits = hits[:fq.limit]
		next := hits[len(hits)-1].AttachmentID
		page.NextCursor = &next
	}
	if page.Items, err = s.messages.FileItems(ctx, hits); err != nil {
		return nil, err
	}
	return page, nil
}

// add puts a sent channel file the caller can read (not a DM's) into the album.
func (s *Service) add(ctx context.Context, userID, albumID, attachmentID id.ID) error {
	if _, _, err := s.editable(ctx, userID, albumID); err != nil {
		return err
	}
	scope, err := s.channels.Readable(ctx, userID)
	if err != nil {
		return err
	}
	var ok bool
	err = s.db.QueryRow(ctx, `SELECT EXISTS (SELECT 1 FROM attachments t JOIN channels c ON c.id = t.channel_id
		JOIN messages m ON m.id = t.message_id
		WHERE t.id = $1 AND NOT m.deleted AND c.type <> 'dm'
			AND (t.channel_id = ANY($2) OR (c.type = 'thread' AND c.parent_channel_id = ANY($2))))`,
		attachmentID, id.Int64s(scope)).Scan(&ok)
	if err != nil {
		return err
	}
	if !ok {
		return httpx.NotFound("attachment")
	}
	if err := s.checkRoom(ctx, albumID); err != nil {
		return err
	}
	tag, err := s.db.Exec(ctx, `INSERT INTO album_items (album_id, attachment_id, added_by) VALUES ($1, $2, $3) ON CONFLICT DO NOTHING`,
		albumID, attachmentID, userID)
	if err != nil {
		return err
	}
	if tag.RowsAffected() > 0 {
		s.itemsChanged(ctx, albumID)
	}
	return nil
}

// take removes an item: a channel file just leaves the album; an album upload is deleted.
func (s *Service) take(ctx context.Context, userID, albumID, attachmentID id.ID) error {
	if _, _, err := s.editable(ctx, userID, albumID); err != nil {
		return err
	}
	var sum, thumb string
	err := s.db.QueryRow(ctx, `DELETE FROM attachments WHERE id = $1 AND album_id = $2 RETURNING sha256, coalesce(thumb_sha256, '')`,
		attachmentID, albumID).Scan(&sum, &thumb)
	switch {
	case err == nil:
		s.itemsChanged(ctx, albumID)
		return s.files.RemoveBlobs(ctx, []string{sum, thumb})
	case !errors.Is(err, pgx.ErrNoRows):
		return err
	}
	tag, err := s.db.Exec(ctx, `DELETE FROM album_items WHERE album_id = $1 AND attachment_id = $2`, albumID, attachmentID)
	if err != nil {
		return err
	}
	if tag.RowsAffected() == 0 {
		return httpx.NotFound("attachment")
	}
	s.itemsChanged(ctx, albumID)
	return nil
}

	s.itemsChanged(ctx, albumID)

func (s *Service) itemsChanged(ctx context.Context, albumID id.ID) {

func registerItems(api huma.API, s *Service, op func(string, string, string, int) huma.Operation, me func(context.Context) id.ID) {
	huma.Register(api, op("list-album-files", http.MethodGet, "/api/albums/{albumID}/files", 0),
		func(ctx context.Context, in *struct {
			AlbumID  string `path:"albumID"`
			Kind     string `query:"kind" doc:"Comma-separated: image, video, audio, document, archive, other"`
			Uploader string `query:"uploader"`
			Q        string `query:"q" maxLength:"100" doc:"Matches file names and captions"`
			Before   string `query:"before" doc:"next_cursor from the previous page"`
			Limit    int    `query:"limit" minimum:"1" maximum:"100" default:"50"`
		}) (*struct{ Body *messages.FilesPage }, error) {
			aid, err := httpx.ParseID(in.AlbumID, "album")
			if err != nil {
				return nil, err
			}
			fq := filesQuery{text: strings.TrimSpace(in.Q), limit: min(max(in.Limit, 1), 100)}
			if in.Kind != "" {
				for _, k := range strings.Split(in.Kind, ",") {
					k = strings.TrimSpace(k)
					if !fileKinds[k] {
```

```ts
export function sections<T extends { created_at: string }>(items: T[], now: Date): { label: string; items: T[] }[] {
  const out: { label: string; items: T[] }[] = [];
  for (const item of items) {

}
```

```ts
/** Albums I own or collaborate on, and other people's albums visible on the server; each keeps the server's order. */
export function groupAlbums(albums: Album[]): { yours: Album[]; server: Album[] } {
  return {
    yours: albums.filter((a) => a.role === "owner" || a.role === "collaborator"),
    server: albums.filter((a) => a.role !== "owner" && a.role !== "collaborator"),
  };
}

/** The albums I can see (with contains, whether each holds that file), kept fresh by album events. */
export function useAlbums(contains?: string) {
  const [albums, setAlbums] = useState<Album[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string>();
  const reload = useCallback(() => {
    setError(undefined);
    client.albums(contains).then(setAlbums, (e) => setError(errorText(e))).finally(() => setLoading(false));
  }, [contains]);
  useEffect(reload, [reload]);
  useEffect(() => client.onEvent((type) => {
    if (type === "ALBUM_UPDATE" || type === "ALBUM_DELETE" || type === "ALBUM_ITEMS_UPDATE") reload();
  }), [reload]);
  return { albums, loading, error, reload };
}
```
